# Runtime dependencies for the macOS build.
#
# Differences from requirements.txt:
#   - sounddevice / colorama dropped: only the CLI (main.py) used them, and the
#     macOS build ships the web UI only. Audio is captured by the browser.
#   - librosa / soxr dropped: they existed to resample for the Whisper pipeline,
#     which is gone. ChunkFormer reads the audio file itself.
#   - transformers dropped along with the PhoWhisper engine. Collecting it cost
#     several hundred megabytes of model architectures the app never used.
#   - jinja2 dropped: the app serves templates/index.html directly.
#
# torch is NOT pinned here because the correct pin is architecture-dependent —
# see packaging/build_macos.sh. PyTorch stopped shipping macOS x86_64 wheels
# after 2.2.2, so Intel Macs are stuck on that release.

numpy>=1.24.0

# Speech to text, running locally. Conformer/CTC: one forward pass instead of
# Whisper's token-by-token decoding, so it is markedly faster and cannot
# hallucinate words out of silence.
# NOTE: unpinned for now. Pin to whatever the first green build installs — an
# unpinned major version is exactly what broke the transformers step earlier.
chunkformer

# Writes the meeting minutes and runs topic tracking.
google-genai>=1.0.0
# Still needed for the optional OpenAI/Groq transcription providers.
openai>=1.0.0

python-dotenv>=1.0.0
fastapi>=0.110.0
uvicorn[standard]>=0.27.0
python-multipart>=0.0.6

# Native window around the local server. On macOS this is WKWebView — the same
# WebKit that Safari uses, already part of the OS — so it costs a wrapper and
# the PyObjC bridge rather than a browser engine. pip pulls pyobjc-core and the
# Cocoa/WebKit/Quartz/Security/UniformTypeIdentifiers frameworks along with it
# through pywebview's own darwin markers, so none of those are listed here.
pywebview>=6.0

# Audio file IO. Kept because chunkformer's loader is likely to reach for it.
soundfile>=0.12.0

# Fetches the model at build time (packaging/fetch_model.py).
huggingface_hub
